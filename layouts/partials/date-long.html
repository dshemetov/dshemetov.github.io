{{- /* Formats a date like "Friday, 17th October 2026". */ -}} {{- $day := .Day -}} {{- $suffix := "th" -}} {{- if not (in (slice 11 12 13) $day) -}} {{- $last := mod $day 10 -}} {{- if eq $last 1 }}{{ $suffix = "st" }}{{ else if eq $last 2 }}{{ $suffix = "nd" }}{{ else if eq $last 3 }}{{ $suffix = "rd" }}{{ end -}} {{- end -}} {{ .Format "Monday" }}, {{ $day }}{{ $suffix }} {{ .Format "January 2006" }}
